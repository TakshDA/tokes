create table if not exists public.live_shifts ( user_id uuid primary key references auth.users(id) on delete cascade, data jsonb not null, updated_at timestamptz not null default now() ); alter table public.live_shifts enable row level security; create policy "own_live_select" on public.live_shifts for select using (auth.uid() = user_id); create policy "own_live_insert" on public.live_shifts for insert with check (auth.uid() = user_id); create policy "own_live_update" on public.live_shifts for update using (auth.uid() = user_id); create policy "own_live_delete" on public.live_shifts for delete using (auth.uid() = user_id);
